// 1.3 · IDENTITY LOCK MECHANISM

วิธีการ Lock · Unlock · Drift Protection

🔒 LOCK
1. Owner / BAPG Authorize
2. Freeze DNA + 7 Libraries
3. Generate SHA-256
4. Mark locked=true
5. Pipeline จะใช้เฉพาะนี้ต่อไป
🔓 UNLOCK (RARE)
1. Request Reason
2. Owner + AI Guardian (SERREZ) approve
3. Bump Major Version (x.0.0)
4. Re-Lock + New Hash
5. Old Render อ้างเก่า · ใหม่อ้างใหม่
🛡 DRIFT PROTECTION
ทุก Render → QC วัด Identity Similarity กับ DNA Reference Set
• < 3% → PASS
• 3–5% → WARNING + Delta Refine
• > 5% → AUTO-REJECT + Log

FUNCTION buildPrompt(characterId, modules, negative, params) { // 1. ดึง DNA ตาม ID · ตรวจ Lock + Hash const dna = await Character.getLocked(characterId); if (!dna.locked) throw "DNA_NOT_LOCKED"; // 2. เรียงโมดูลตามลำดับมาตรฐาน · 01 = IDENTITY เสมอ const ordered = [ { id:'identity', text:dna.prompt_fragment, locked:true, weight:dna.weight }, ...modules.sort((a,b)=>ORDER[a.id]-ORDER[b.id]) ]; // 3. ตัดโมดูลที่ขัดแย้งกับ DNA (Drift Guard) ordered.forEach(m => DriftGuard.validate(m, dna)); // 4. ประกอบ = น้ำหนักแปรเป็น Position + Repetition const positive = WeightedAssembler.run(ordered); const negative = NegativeAssembler.run(negative); // 5. Generate Hash + Version · เก็บเป็นประวัติ const hash = sha256(positive + "|" + negative + dna.sha256); PromptVersion.create({ characterId, modules, hash, params }); return { positive, negative, hash, dnaVersion:dna.version }; }
